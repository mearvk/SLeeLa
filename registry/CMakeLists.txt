cmake_minimum_required(VERSION 3.16)
project(SLeeLaRegistry3 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(sleela_registry_core STATIC sleela_registry.c v3/sleela_registry_v3.c)
target_include_directories(sleela_registry_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/v3)
add_library(sleela_registry_cpp INTERFACE)
target_include_directories(sleela_registry_cpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/v3)
target_link_libraries(sleela_registry_cpp INTERFACE sleela_registry_core)
add_executable(sleela-registry3 v3/sleela-registry3.c)
target_link_libraries(sleela-registry3 PRIVATE sleela_registry_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_registry_v3 tests/test_registry_v3.c)
 target_link_libraries(test_registry_v3 PRIVATE sleela_registry_core)
 add_test(NAME registry-v3-records COMMAND test_registry_v3)
 add_executable(test_registry_v3_cpp tests/test_registry_v3_cpp.cpp)
 target_link_libraries(test_registry_v3_cpp PRIVATE sleela_registry_cpp)
 add_test(NAME registry-v3-cpp-facade COMMAND test_registry_v3_cpp)
endif()
